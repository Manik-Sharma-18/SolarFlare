     ```python
     def seed_everything(seed: int):
         """Set seeds for reproducibility across torch, numpy, and python random."""
         torch.manual_seed(seed)
         if torch.cuda.is_available():
             torch.cuda.manual_seed_all(seed)
         np.random.seed(seed)
         random.seed(seed)
     ```
